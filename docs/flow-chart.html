✓ After
function timingSafeEqual(a, b) {
  if (a.length !== b.length) return false;
  let diff = 0;
  for (let i = 0; i < a.length; i++)
    diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return diff === 0;
}

export default async function handler(req, res) {
  const expected = process.env.TRACKER_SECRET;
  if (expected) {
    const auth = String(req.headers.authorization);
    const presented = auth.startsWith('Bearer ')
      ? auth.slice(7) : '';
    if (!presented || !timingSafeEqual(presented, expected)) {
      res.status(401).json({ error: 'unauthorized' });
      return;
    }
  }
  // … tiếp tục như cũ …
}

function ensureTempRoot() {
  fs.mkdirSync(TEMP_ROOT, { recursive: true });
  return TEMP_ROOT;

export function sweepLeakedPdfs() {

    fs.rmSync(path.join(TEMP_ROOT, name), { recursive: true, force: true });
  }

In-session cleanup vẫn chạy như cũ (finally block). Hai lớp phòng ngự = không còn PDF chứa bảng lương rò rỉ.

✓ Gap 13 — Reject Month/Year không hợp lệ

File: src/lib/autoMapping.ts
❌ Before
export function parsePeriod(raw) {
  const s = String(raw).trim();
  let m = s.match(/^(\d{1,2})\/(\d{4})$/);
  if (m) return { month: m[1].padStart(2, '0'), year: m[2] };
  m = s.match(/^(\d{4})-(\d{1,2})$/);
  if (m) return { month: m[2].padStart(2, '0'), year: m[1] };
  return null;
  // ⚠️ "13/2026" pass → gửi email "tháng 13"
  // "00/2026" pass → subject kỳ quặc
  // "05/20260" pass → year ảo
}
✓ After
export function parsePeriod(raw) {
  // … match regex như cũ, lấy monthStr + yearStr …
  if (!monthStr || !yearStr) return null;
  const month = Number(monthStr);
  const year = Number(yearStr);
  if (!Number.isInteger(month) || month < 1 || month > 12)
    return null;
  // Range hợp lý — reject 202, 20260, 1999, v.v.
  if (!Number.isInteger(year) || year < 2000 || year > 2100)
    return null;
  return { month: String(month).padStart(2, '0'), year: String(year) };
}
Tại sao: Regex chỉ check format, không check range. Giờ validate thêm month ∈ [1, 12] và year ∈ [2000, 2100]. Invalid → trả null, App.tsx fallback sang tháng hiện tại (new Date()). User không phải gõ đúng format — app dùng kỳ hiện tại nếu Excel sai.

  }
  throw lastError;
}
